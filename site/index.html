1-Minute Setup

One line of code. Zero headache.

Mount the Pulse middleware in your favourite backend framework and open /pulse.

server.js / main.py / main.go 
import express from 'express';
import { Monitor, expressPulseMiddleware } from '@alaminislam34/server-monitor';

const app = express();
const monitor = new Monitor({ openApiSpecUrl: '/openapi.json' });

// 1-Line Pulse Integration (Serves /pulse and monitors all traffic)
app.use(expressPulseMiddleware(monitor));

app.listen(3000, () => console.log('Pulse Cockpit: http://localhost:3000/pulse'));
from fastapi import FastAPI
from pulse_monitor import PulseMiddleware

app = FastAPI()

# 1-Line Pulse ASGI Integration (Just like Swagger UI)
app.add_middleware(PulseMiddleware, endpoint="/pulse", open_api_spec_url="/openapi.json")

@app.get("/api/v1/users")
def get_users():
    return [{"id": 1, "name": "Alice"}]
package main

import (
    "net/http"
    "github.com/alaminislam34/pulse-monitor/sdk/go/pulse"
)

func main() {
    m := pulse.NewMonitor(pulse.Config{BasePath: "/pulse"})
    mux := http.NewServeMux()

    // 1-Line Pulse Middleware (Serves /pulse and monitors all routes)
    http.ListenAndServe(":8080", pulse.HTTPMiddleware(m)(mux))
}
import { NestFactory } from '@nestjs/core';
import { Monitor, expressPulseMiddleware } from '@alaminislam34/server-monitor';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  const monitor = new Monitor({ openApiSpecUrl: '/api-json' });

  app.use(expressPulseMiddleware(monitor));
  await app.listen(3000);
}
bootstrap();

Frequently Asked Questions

Everything you need to know.

How is Pulse Monitor different from Swagger UI?

Swagger UI is a static documentation reader: it tells you what your API is supposed to do, but has no idea what real traffic is actually hitting your server. Pulse Monitor reads your OpenAPI specification, but it also captures live traffic in real time and highlights contract drift (e.g. missing required fields or undocumented 500 errors) as they happen.

Does Pulse Monitor send my API data to any external cloud?

No. Pulse Monitor operates with strict zero-cloud egress. All telemetry, request ring buffers, and contract comparisons happen locally in your process memory and are served directly from your application's /pulse route. Sensitive headers and request/response secrets are scrubbed automatically.

How much runtime overhead does Pulse Monitor introduce?

Sub-millisecond (< 0.4ms). Pulse uses a bounded circular ring buffer with mutex-guarded memory pooling. It never writes to disk and never grows memory unbounded. When the buffer reaches capacity, the oldest records are dropped in FIFO order.

Can I use Pulse Monitor across multiple backend languages?

Yes! Pulse Monitor is built on an open wire protocol specification (SPEC.md). We currently provide production SDKs for Node.js (Express & NestJS), Python (FastAPI & Starlette), and Go (net/http & Gin).
