Core Capabilities

Built for Modern API Engineers

Everything you need to observe, debug, and enforce API contracts during development and staging.

⚡

1-Line Universal Adapter

Drop a single middleware into Express, NestJS, or FastAPI. Automatically opens a dedicated dashboard on /pulse with zero setup.

🛡️

Intrusion Detection Watchdog

Built-in heuristic scanner inspects SQL Injection, Cross-Site Scripting (XSS), and Path Traversal signatures in real time.

📖

OpenAPI Schema Drift Detector

Ingests /openapi.json and validates live requests against the contract. Flags missing fields and undocumented endpoints in red.

🔐

Auto Credential Sanitization

Zero risk of leaking secrets. Authorization headers, session cookies, JWTs, and passwords are automatically masked with [REDACTED].

📡

Live Server-Sent Events (SSE)

Streams telemetry straight from memory to the browser. See requests, latency spikes, and threats with zero manual page refreshes.

☁️

100% Local-First Privacy

Zero cloud accounts, zero third-party telemetry egress. Uses pre-allocated circular ring buffers to maintain zero-latency overhead.

60-Second Integration

Drop into Your Favorite Framework

Select your language and framework below to see how effortless setup is. 

import express from 'express';
import { Monitor, expressPulseMiddleware } from 'pulse-monitor';

const app = express();
const monitor = new Monitor({ openApiSpecUrl: '/openapi.json' });

// 1-Line Pulse Integration
app.use(expressPulseMiddleware(monitor));

app.listen(3000, () => {
  console.log('Dashboard: http://localhost:3000/pulse');
});
from fastapi import FastAPI
from pulse_monitor import PulseMiddleware

app = FastAPI()

# 1-Line Pulse Integration (Just like Swagger UI!)
app.add_middleware(PulseMiddleware, endpoint="/pulse", open_api_spec_url="/openapi.json")

@app.get("/api/v1/users")
def get_users():
    return [{"id": 1, "name": "Alice"}]
package main

import (
    "net/http"
    "github.com/alaminislam34/pulse-monitor/sdk/go/pulse"
)

func main() {
    m := pulse.NewMonitor(pulse.Config{BasePath: "/pulse"})
    mux := http.NewServeMux()

    // 1-Line Pulse Middleware (Serves /pulse and monitors all routes)
    http.ListenAndServe(":8080", pulse.HTTPMiddleware(m)(mux))
}
import { NestFactory } from '@nestjs/core';
import { Monitor, expressPulseMiddleware } from 'pulse-monitor';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  const monitor = new Monitor();

  app.use(expressPulseMiddleware(monitor));
  await app.listen(3000);
}
bootstrap();
